// --- CART FUNCTIONS (global) ---
window.closeCart = function() {
    jQuery('#cartOverlay, #miniCart').removeClass('open');
    document.body.style.overflow = '';
};
window.openCart = function() {
    jQuery('#cartOverlay, #miniCart').addClass('open');
    document.body.style.overflow = 'hidden';
    updateCartUI();
};
window.toggleCart = function() {
    if (jQuery('#cartOverlay').hasClass('open')) closeCart();
    else openCart();
};

function updateCartUI() {
    jQuery.get(yenolabel_ajax.ajax_url, { action: 'get_cart_count', nonce: yenolabel_ajax.nonce }, function(count) {
        jQuery('#cart-count, #mini-cart-count').text(count);
    });
    jQuery.get(yenolabel_ajax.ajax_url, { action: 'get_cart_items', nonce: yenolabel_ajax.nonce }, function(html) {
        jQuery('#cart-items').html(html);
    });
    jQuery.get(yenolabel_ajax.ajax_url, { action: 'get_cart_subtotal', nonce: yenolabel_ajax.nonce }, function(html) {
        jQuery('#cart-total').html(html);
    });
}

window.updateCartItem = function(key, qty) {
    jQuery.post(yenolabel_ajax.ajax_url, {
        action: 'update_cart_item',
        key: key,
        qty: qty,
        nonce: yenolabel_ajax.nonce
    }, function() { updateCartUI(); });
};
window.removeCartItem = function(key) {
    jQuery.post(yenolabel_ajax.ajax_url, {
        action: 'remove_cart_item',
        key: key,
        nonce: yenolabel_ajax.nonce
    }, function() { updateCartUI(); });
};

// --- SEARCH ---
window.openSearch = function() {
    document.getElementById('searchOverlay').classList.add('open');
    document.getElementById('searchInput').focus();
};
window.closeSearch = function() {
    document.getElementById('searchOverlay').classList.remove('open');

window.performSearch = function() {
    var q = document.getElementById('searchInput').value.trim();
    if (q.length < 2) { document.getElementById('searchResults').innerHTML = ''; return; }
    jQuery.get(yenolabel_ajax.ajax_url, { action: 'yeno_search', q: q, nonce: yenolabel_ajax.nonce }, function(data) {
        var html = '';
        data.forEach(function(item) {
            html += '
';
            html += '';
            html += ''+item.name+'
';
        });
        document.getElementById('searchResults').innerHTML = html || '
No results
';
    });
};

window.toggleTheme = function() {
    var html = document.documentElement;
    var current = html.getAttribute('data-theme');
    var next = current === 'dark' ? 'light' : 'dark';
    html.setAttribute('data-theme', next);
    localStorage.setItem('theme', next);
};
